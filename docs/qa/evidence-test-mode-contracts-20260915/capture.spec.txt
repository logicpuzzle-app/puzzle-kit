import { test, expect } from '../e2e/fixtures';

test('Embedded modes expose shortcuts, reset submodes and cycle selection', async ({ page }) => {
  await page.goto('/embedded.html');
  const panel = page.locator('.mode-panel');
  const surface = panel.getByTitle('Surface (S)', { exact: true });
  await expect(surface).toBeVisible();
  await panel.getByTitle('Line (L)', { exact: true }).click();
  await expect(panel.getByRole('button', { name: 'Line', exact: true }))
    .toHaveCSS('background-color', 'rgb(227, 242, 253)');
  const free = panel.getByRole('button', { name: 'Freeline', exact: true });
  await free.click();
  await expect(free).toHaveCSS('background-color', 'rgb(227, 242, 253)');
  await panel.getByTitle('Wall (W)', { exact: true }).click();
  await expect(panel.getByRole('button', { name: 'Wall', exact: true }))
    .toHaveCSS('background-color', 'rgb(227, 242, 253)');
  await expect(free).toHaveCount(0);
  await surface.click();
  await surface.click();
  await expect(panel.getByRole('button', { name: 'Dot', exact: true }))
    .toHaveCSS('background-color', 'rgb(227, 242, 253)');
  await expect(panel.getByRole('button', { name: 'Multicolor', exact: true })).toBeVisible();
  await page.evaluate(() => Promise.all(document.getAnimations().map(animation => animation.finished)));
});
