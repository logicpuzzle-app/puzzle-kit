import { test, expect } from '../e2e/fixtures';

for (const [kind, label] of [['triangle', '△ Triangle'], ['pyramid', '▲ Pyramid']] as const) {
  test(`New ${kind} board renders the selected geometry`, async ({ page }) => {
    await page.goto('/master');
    await page.getByRole('button', { name: 'File', exact: true }).click();
    await page.getByText('New', { exact: true }).click();
    const dialog = page.getByRole('heading', { name: 'New', exact: true }).locator('..');
    await dialog.getByRole('button', { name: label, exact: true }).click();
    await dialog.getByRole('spinbutton').nth(0).fill('4');
    await dialog.getByRole('spinbutton').nth(1).fill('4');
    await dialog.getByRole('button', { name: 'Create', exact: true }).click();
    await expect(page.locator('#puzzle-canvas')).toBeVisible();
    await expect.poll(() => page.evaluate(async () => {
      const path = '/src/store/puzzleStore.ts';
      const { usePuzzleStore } = await import(path);
      const { grid } = usePuzzleStore.getState();
      return { type: grid.gridType, rows: grid.rows, cols: grid.cols };
    })).toEqual({ type: kind, rows: 4, cols: 4 });
    await expect(page.locator('#puzzle-canvas .topology-grid-layer polygon').first()).toBeVisible();
    await page.evaluate(() => Promise.all(document.getAnimations().map(animation => animation.finished)));
  });
}
